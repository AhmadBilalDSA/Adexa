from flask import Flask, render_template_string, request import os import sys sys.path.append(os.path.abspath(os.path.join(os.path.dirname(__file__), ".."))) from debugger.gdb_runner import run_gdb from web_engine.pipeline import analyze_web_request app = Flask(__name__) # --- Simple HTML UI Template --- HTML = """ ADEXA Interface

ADEXA Web Interface

Crash Analysis

Web Request Analyzer

{% if result %}

Result

{{ result }}
{% endif %} """ @app.route("/", methods=["GET"]) def index(): return render_template_string(HTML) @app.route("/crash", methods=["POST"]) def crash(): binary = request.form.get("binary") output = run_gdb(binary) # triggers crash engine return render_template_string(HTML, result=output) @app.route("/web", methods=["POST"]) def web(): raw = request.form.get("raw_request") result = analyze_web_request(raw) return render_template_string(HTML, result=result) if __name__ == "__main__": app.run(host="127.0.0.1", port=5000, debug=True)
